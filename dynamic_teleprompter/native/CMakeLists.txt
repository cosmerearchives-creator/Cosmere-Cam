cmake_minimum_required(VERSION 3.16)
project(dynamic_teleprompter_native LANGUAGES CXX)
add_library(dt_engine SHARED engine.cpp)
target_compile_features(dt_engine PRIVATE cxx_std_17)
set_target_properties(dt_engine PROPERTIES PREFIX "" OUTPUT_NAME "dt_engine")
if(MSVC)
    target_compile_options(dt_engine PRIVATE /O2 /W4)
else()
    target_compile_options(dt_engine PRIVATE -O3 -Wall -Wextra)
endif()
if(WIN32)
    add_executable(cosmere_vst2_host vst2/host.cpp)
    target_compile_features(cosmere_vst2_host PRIVATE cxx_std_17)
    if(MSVC)
        target_compile_options(cosmere_vst2_host PRIVATE /utf-8)
    endif()
    target_link_libraries(cosmere_vst2_host PRIVATE user32)
endif()
